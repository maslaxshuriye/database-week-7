-- ==================================================
-- STEP 1: CREATE THE AUDIT LOG AND TRIGGER
-- ==================================================

CREATE TABLE IF NOT EXISTS audit_log (
id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
tbl TEXT NOT NULL,
op TEXT NOT NULL,
old_row JSONB,
new_row JSONB,
changed_by TEXT DEFAULT current_user,
at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE OR REPLACE FUNCTION audit()
RETURNS TRIGGER AS $$
BEGIN
INSERT INTO audit_log (tbl, op, old_row, new_row)
VALUES (
TG_TABLE_NAME,
TG_OP,
CASE
WHEN TG_OP IN ('UPDATE', 'DELETE')
THEN to_jsonb(OLD)
ELSE NULL
END,
CASE
WHEN TG_OP IN ('INSERT', 'UPDATE')
THEN to_jsonb(NEW)
ELSE NULL
END
);

```
RETURN CASE
    WHEN TG_OP = 'DELETE' THEN OLD
    ELSE NEW
END;
```

END;

$$$LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_audit ON students;

CREATE TRIGGER trg_audit
AFTER INSERT OR UPDATE OR DELETE
ON students
FOR EACH ROW
EXECUTE FUNCTION audit();

-- Update a student's name.
UPDATE students
SET name = 'Amina Yusuf'
WHERE id = 1;

-- Delete the student with ID 3.

WHERE id = 3;

-- View the recorded changes.
SELECT
    id,
    tbl AS table_name,
    op AS operation,
    old_row ->> 'name' AS previous_name,
    new_row ->> 'name' AS updated_name,
    changed_by,
    at AS changed_at
FROM audit_log
ORDER BY id DESC;

-- STEP 3: CREATE AND QUERY A CATEGORY TREE
-- ==================================================

CREATE TABLE IF NOT EXISTS categories (
    id SERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    parent_id INTEGER REFERENCES categories(id)
);

-- Add example categories without duplicating these IDs.
INSERT INTO categories (id, name, parent_id)
VALUES
    (1, 'Education', NULL),
    (2, 'Books', 1),
    (3, 'Science Books', 2),
    (4, 'Online Courses', 1)
ON CONFLICT (id) DO NOTHING;

-- Synchronize the category ID sequence.
SELECT setval(
    pg_get_serial_sequence('categories', 'id'),
    GREATEST(COALESCE((SELECT MAX(id) FROM categories), 1), 1),
    EXISTS (SELECT 1 FROM categories)
);

-- Display categories in hierarchical order.
WITH RECURSIVE category_tree AS (
    SELECT
        id,
        name,
        parent_id,
        0 AS depth
    FROM categories
    WHERE parent_id IS NULL

    UNION ALL

    SELECT
        child.id,
        child.name,
        child.parent_id,
        parent.depth + 1
    FROM categories AS child
    JOIN category_tree AS parent
        ON child.parent_id = parent.id
)
SELECT
    id,
    repeat('  ', depth) || name AS category,
    parent_id,
    depth
FROM category_tree
ORDER BY id;

-- STEP 5: CREATE ROLES AND GRANT PERMISSIONS
-- ==================================================

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_roles WHERE rolname = 'app_read'
    ) THEN
        CREATE ROLE app_read;
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_roles WHERE rolname = 'app_write'
    ) THEN
        CREATE ROLE app_write;
    END IF;
END
$$;

-- Permit database connections.
GRANT CONNECT ON DATABASE bootcamp TO app_read, app_write;

-- Permit access to the public schema.
GRANT USAGE ON SCHEMA public TO app_read, app_write;

-- Read-only access.
GRANT SELECT ON ALL TABLES IN SCHEMA public TO app_read;

-- Read and write access.
GRANT SELECT, INSERT, UPDATE, DELETE
ON ALL TABLES IN SCHEMA public TO app_write;

-- Sequence access required for generated IDs.
GRANT USAGE, SELECT
ON ALL SEQUENCES IN SCHEMA public TO app_write;

-- Default privileges for future tables.
ALTER DEFAULT PRIVILEGES IN SCHEMA public
GRANT SELECT ON TABLES TO app_read;

ALTER DEFAULT PRIVILEGES IN SCHEMA public
GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO app_write;

-- Default privileges for future sequences.
ALTER DEFAULT PRIVILEGES IN SCHEMA public
GRANT USAGE, SELECT ON SEQUENCES TO app_write;

-- Create an API login if it does not already exist.
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_roles WHERE rolname = 'api'
    ) THEN
        CREATE ROLE api LOGIN PASSWORD 'ChangeThis_StrongPassword_2026!';
    END IF;
END
$$;

GRANT app_write TO api;
