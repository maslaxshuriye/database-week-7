-- Create the audit log table.

id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
tbl TEXT NOT NULL,
op TEXT NOT NULL,
old_row JSONB,
new_row JSONB,
changed_by TEXT DEFAULT current_user,
at TIMESTAMPTZ DEFAULT now()
);
